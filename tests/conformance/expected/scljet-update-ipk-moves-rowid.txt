-- the bug report's first form: WHERE on the IPK column itself --
ok: UPDATE emp SET id = 5 WHERE id = 1
  after: 5|5|ann
-- the second form: WHERE on another column --
ok: UPDATE emp SET id = 9 WHERE name = 'ann'
  after: 9|9|ann
-- a non-IPK update leaves the rowid alone --
ok: UPDATE emp SET name = 'bea' WHERE id = 9
  after: 9|9|bea
-- moving onto an occupied rowid is refused, and changes nothing --
ok: INSERT INTO emp VALUES (10, 'cid')
refused: UPDATE emp SET id = 10 WHERE id = 9 -- UNIQUE constraint failed: rowid 10 already exists
  after: 9|9|bea ; 10|10|cid
-- a non-integer target is refused rather than coerced --
refused: UPDATE emp SET id = 'x' WHERE id = 9 -- datatype mismatch: INTEGER PRIMARY KEY must be assigned an integer
  after: 9|9|bea ; 10|10|cid
-- indexed: the moved row must change POSITION, not just key --
ok: UPDATE emp SET id = 9 WHERE id = 1
  after: 2|2|bob ; 9|9|ann
-- indexed: a collision is refused here too, with the same wording --
refused: UPDATE emp SET id = 2 WHERE id = 1 -- UNIQUE constraint failed: rowid 2 already exists
  after: 1|1|ann ; 2|2|bob
